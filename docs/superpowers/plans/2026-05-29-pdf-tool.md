package fan.summer.buildintool.pdftool.converter;

import fan.summer.api.log.LoggerFactory;
import fan.summer.api.log.PluginLogger;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Optional;
import java.util.concurrent.atomic.AtomicReference;

public class OfficeDetector {

    private static final PluginLogger log = LoggerFactory.getLogger(OfficeDetector.class);
    private static final AtomicReference<DetectedBackend> cached = new AtomicReference<>();

    public enum BackendType { WPS, LIBRE_OFFICE, MS_WORD }

    public record DetectedBackend(BackendType type, String executablePath, String displayName) {}

    public static Optional<DetectedBackend> detect() {
        DetectedBackend existing = cached.get();
        if (existing != null) return Optional.of(existing);

        String os = System.getProperty("os.name", "").toLowerCase();
        boolean isMac = os.contains("mac") || os.contains("darwin");
        boolean isWin = os.contains("win");

        Optional<DetectedBackend> result;
        if (isMac) result = detectMac();
        else if (isWin) result = detectWindows();
        else result = detectLinux();

        result.ifPresent(cached::set);
        if (result.isEmpty()) log.warn("No Office backend detected");
        else log.info("Detected Office backend: {}", result.get().displayName);
        return result;
    }

    public static void clearCache() { cached.set(null); }

    // --- macOS ---

    private static Optional<DetectedBackend> detectMac() {
        return tryDetect(
            // WPS
            () -> checkMacApp("wpsoffice.app", "wps").map(p -> new DetectedBackend(BackendType.WPS, p, "WPS Office")),
            // LibreOffice
            () -> checkMacApp("LibreOffice.app", "soffice").map(p -> new DetectedBackend(BackendType.LIBRE_OFFICE, p, "LibreOffice")),
            // MS Word (limited on macOS — no headless, but detect for completeness)
            () -> checkMacApp("Microsoft Word.app", "Microsoft Word").map(p -> new DetectedBackend(BackendType.MS_WORD, p, "Microsoft Word"))
        );
    }

    private static Optional<String> checkMacApp(String appName, String binary) {
        Path[] paths = {
            Path.of("/Applications").resolve(appName).resolve("Contents/MacOS").resolve(binary),
            Path.of(System.getProperty("user.home"), "Applications").resolve(appName).resolve("Contents/MacOS").resolve(binary)
        };
        for (Path p : paths) {
            if (Files.isExecutable(p)) return Optional.of(p.toString());
        }
        return Optional.empty();
    }

    // --- Windows ---

    private static Optional<DetectedBackend> detectWindows() {
        return tryDetect(
            // WPS
            () -> detectWindowsWps(),
            // LibreOffice
            () -> detectWindowsLibreOffice(),
            // MS Word
            () -> detectWindowsWord()
        );
    }

    private static Optional<DetectedBackend> detectWindowsWps() {
        String programFiles = System.getenv("ProgramFiles");
        String programFilesX86 = System.getenv("ProgramFiles(X86)");
        String localAppData = System.getenv("LOCALAPPDATA");
        String userHome = System.getProperty("user.home");

        String[] candidates = {
            programFiles + "\\Kingsoft\\WPS Office\\",
            programFilesX86 + "\\Kingsoft\\WPS Office\\",
            localAppData + "\\Kingsoft\\WPS Office\\",
            userHome + "\\AppData\\Local\\Kingsoft\\WPS Office\\"
        };
        for (String base : candidates) {
            if (base == null) continue;
            File dir = new File(base);
            if (!dir.isDirectory()) continue;
            // Look for office6/wps.exe inside version subdirectories
            File[] subs = dir.listFiles(File::isDirectory);
            if (subs != null) {
                for (File sub : subs) {
                    File wps = new File(sub, "office6\\wps.exe");
                    if (wps.isFile()) return Optional.of(new DetectedBackend(BackendType.WPS, wps.getAbsolutePath(), "WPS Office"));
                }
            }
        }
        return Optional.empty();
    }

    private static Optional<DetectedBackend> detectWindowsLibreOffice() {
        String programFiles = System.getenv("ProgramFiles");
        String programFilesX86 = System.getenv("ProgramFiles(X86)");
        String[] candidates = {
            programFiles + "\\LibreOffice\\program\\soffice.exe",
            programFilesX86 + "\\LibreOffice\\program\\soffice.exe"
        };
        for (String path : candidates) {
            if (path == null) continue;
            File f = new File(path);
            if (f.isFile()) return Optional.of(new DetectedBackend(BackendType.LIBRE_OFFICE, f.getAbsolutePath(), "LibreOffice"));
        }
        return Optional.empty();
    }

    private static Optional<DetectedBackend> detectWindowsWord() {
        String programFiles = System.getenv("ProgramFiles");
        String programFilesX86 = System.getenv("ProgramFiles(X86)");
        String[] candidates = {
            programFiles + "\\Microsoft Office\\root\\Office16\\WINWORD.EXE",
            programFilesX86 + "\\Microsoft Office\\root\\Office16\\WINWORD.EXE",
            programFiles + "\\Microsoft Office 15\\root\\office15\\WINWORD.EXE"
        };
        for (String path : candidates) {
            if (path == null) continue;
            File f = new File(path);
            if (f.isFile()) return Optional.of(new DetectedBackend(BackendType.MS_WORD, f.getAbsolutePath(), "Microsoft Word"));
        }
        return Optional.empty();
    }

    // --- Linux ---

    private static Optional<DetectedBackend> detectLinux() {
        return tryDetect(
            () -> checkLinuxBinary("/opt/kingsoft/wps-office/office6/wps", BackendType.WPS, "WPS Office"),
            () -> checkLinuxWhich("wps", BackendType.WPS, "WPS Office"),
            () -> checkLinuxWhich("libreoffice", BackendType.LIBRE_OFFICE, "LibreOffice"),
            () -> checkLinuxWhich("soffice", BackendType.LIBRE_OFFICE, "LibreOffice")
        );
    }

    private static Optional<DetectedBackend> checkLinuxBinary(String path, BackendType type, String name) {
        File f = new File(path);
        return f.isFile() ? Optional.of(new DetectedBackend(type, f.getAbsolutePath(), name)) : Optional.empty();
    }

    private static Optional<DetectedBackend> checkLinuxWhich(String cmd, BackendType type, String name) {
        try {
            Process p = new ProcessBuilder("which", cmd).redirectErrorStream(true).start();
            String output = new String(p.getInputStream().readAllBytes()).trim();
            if (p.waitFor() == 0 && !output.isEmpty()) return Optional.of(new DetectedBackend(type, output, name));
        } catch (Exception ignored) {}
        return Optional.empty();
    }

    // --- helper ---

    @SafeVarargs
    private static Optional<DetectedBackend> tryDetect(java.util.function.Supplier<Optional<DetectedBackend>>... checks) {
        for (var check : checks) {
            try {
                Optional<DetectedBackend> result = check.get();
                if (result.isPresent()) return result;
            } catch (Exception e) {
                log.debug("Office detection check failed: {}", e.getMessage());
            }
        }
        return Optional.empty();
    }
}

package fan.summer.buildintool.pdftool.converter;

import fan.summer.api.log.LoggerFactory;
import fan.summer.api.log.PluginLogger;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.concurrent.TimeUnit;

public class WpsConverter implements DocumentConverter {

    private static final PluginLogger log = LoggerFactory.getLogger(WpsConverter.class);
    private static final int TIMEOUT_MINUTES = 10;

    private final String wpsExecutable;

    public WpsConverter(String wpsExecutable) {
        this.wpsExecutable = wpsExecutable;
    }

    @Override
    public void convert(Path pdfPath, Path outputPath) throws Exception {
        String os = System.getProperty("os.name", "").toLowerCase();
        ProcessBuilder pb;
        if (os.contains("win")) {
            // Windows: direct executable with headless flag
            pb = new ProcessBuilder(
                wpsExecutable,
                "--headless",
                "--convert-to", "docx",
                pdfPath.toString(),
                "--outdir", outputPath.getParent().toString()
            );
        } else {
            // macOS/Linux: use the executable directly
            pb = new ProcessBuilder(
                wpsExecutable,
                "--headless",
                "--convert-to", "docx",
                pdfPath.toString(),
                "--outdir", outputPath.getParent().toString()
            );
        }
        pb.redirectErrorStream(true);
        log.info("Starting WPS conversion: {} -> {}", pdfPath.getFileName(), outputPath.getFileName());

        Process process = pb.start();
        String output = new String(process.getInputStream().readAllBytes());
        boolean finished = process.waitFor(TIMEOUT_MINUTES, TimeUnit.MINUTES);

        if (!finished) {
            process.destroyForcibly();
            throw new IOException("WPS conversion timed out after " + TIMEOUT_MINUTES + " minutes");
        }
        int exitCode = process.exitValue();
        if (exitCode != 0) {
            throw new IOException("WPS conversion failed (exit=" + exitCode + "): " + output.trim());
        }
        if (!Files.exists(outputPath)) {
            throw new IOException("WPS conversion produced no output file: " + outputPath);
        }
        log.info("WPS conversion completed: {}", outputPath.getFileName());
    }
}

package fan.summer.buildintool.pdftool.converter;

import com.documents4j.api.DocumentType;
import com.documents4j.api.IConverter;
import com.documents4j.job.LocalConverter;
import fan.summer.api.log.LoggerFactory;
import fan.summer.api.log.PluginLogger;

import java.io.File;
import java.nio.file.Path;
import java.util.concurrent.TimeUnit;

public class Documents4jConverter implements DocumentConverter {

    private static final PluginLogger log = LoggerFactory.getLogger(Documents4jConverter.class);
    private static final int TIMEOUT_MINUTES = 10;

    @Override
    public void convert(Path pdfPath, Path outputPath) throws Exception {
        log.info("Starting documents4j conversion: {} -> {}", pdfPath.getFileName(), outputPath.getFileName());
        File pdfFile = pdfPath.toFile();
        File docxFile = outputPath.toFile();

        IConverter converter = LocalConverter.builder().build();
        boolean success = converter.convert(pdfFile).as(DocumentType.PDF)
                .to(docxFile).as(DocumentType.DOCX)
                .prioritizeWithOcr(true)
                .execute(TIMEOUT_MINUTES, TimeUnit.MINUTES);

        converter.shutDown();

        if (!success) {
            throw new IOException("documents4j conversion failed for: " + pdfPath.getFileName());
        }
        log.info("documents4j conversion completed: {}", outputPath.getFileName());
    }
}

package fan.summer.buildintool.pdftool.worker;

import fan.summer.api.log.LoggerFactory;
import fan.summer.api.log.PluginLogger;
import javafx.concurrent.Task;
import org.apache.pdfbox.Loader;
import org.apache.pdfbox.pdmodel.PDDocument;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class PdfSplitWorker extends Task<List<Path>> {

    private static final PluginLogger log = LoggerFactory.getLogger(PdfSplitWorker.class);

    private final Path pdfPath;
    private final List<int[]> ranges; // each int[] is [start, end], 1-based inclusive
    private final Path outputDir;

    public PdfSplitWorker(Path pdfPath, List<int[]> ranges, Path outputDir) {
        this.pdfPath = pdfPath;
        this.ranges = ranges;
        this.outputDir = outputDir;
    }

    @Override
    protected List<Path> call() throws Exception {
        Files.createDirectories(outputDir);
        String baseName = com.google.common.io.Files.getNameWithoutExtension(pdfPath.toString());
        // Fallback if Guava is not available:
        String fileName = pdfPath.getFileName().toString();
        int dotIdx = fileName.lastIndexOf('.');
        baseName = dotIdx > 0 ? fileName.substring(0, dotIdx) : fileName;

        List<Path> outputs = new ArrayList<>();

        try (PDDocument doc = Loader.loadPDF(pdfPath.toFile())) {
            int total = doc.getNumberOfPages();
            int processed = 0;

            for (int[] range : ranges) {
                int start = range[0]; // 1-based
                int end = range[1];   // 1-based inclusive

                try (PDDocument splitDoc = new PDDocument()) {
                    for (int p = start; p <= end; p++) {
                        splitDoc.addPage(doc.getPage(p - 1)); // 0-based in PDDocument
                    }
                    String rangeLabel = (start == end) ? "p" + start : "p" + start + "-" + end;
                    Path outPath = resolveUniquePath(outputDir, baseName + "_" + rangeLabel + ".pdf");
                    splitDoc.save(outPath.toFile());
                    outputs.add(outPath);
                    log.info("Split page range {}-{} -> {}", start, end, outPath.getFileName());
                }
                processed++;
                updateProgress(processed, ranges.size());
                updateMessage("Splitting range " + processed + "/" + ranges.size());
            }
        }
        return outputs;
    }

    private Path resolveUniquePath(Path dir, String fileName) {
        Path candidate = dir.resolve(fileName);
        if (!Files.exists(candidate)) return candidate;
        String name = fileName.substring(0, fileName.lastIndexOf('.'));
        String ext = fileName.substring(fileName.lastIndexOf('.'));
        int seq = 1;
        while (Files.exists(dir.resolve(name + "(" + seq + ")" + ext))) seq++;
        return dir.resolve(name + "(" + seq + ")" + ext);
    }
}

package fan.summer.buildintool.pdftool.worker;

import fan.summer.api.log.LoggerFactory;
import fan.summer.api.log.PluginLogger;
import javafx.concurrent.Task;
import org.apache.pdfbox.Loader;
import org.apache.pdfbox.io.IOUtils;
import org.apache.pdfbox.multipdf.PDFMergerUtility;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;

public class PdfMergeWorker extends Task<Path> {

    private static final PluginLogger log = LoggerFactory.getLogger(PdfMergeWorker.class);

    private final List<Path> filePaths;
    private final Path outputPath;

    public PdfMergeWorker(List<Path> filePaths, Path outputPath) {
        this.filePaths = filePaths;
        this.outputPath = outputPath;
    }

    @Override
    protected Path call() throws Exception {
        Files.createDirectories(outputPath.getParent());
        PDFMergerUtility merger = new PDFMergerUtility();
        merger.setDestinationFileName(resolveUniquePath(outputPath).toString());

        int total = filePaths.size();
        for (int i = 0; i < total; i++) {
            merger.addSource(filePaths.get(i).toFile());
            updateProgress(i + 1, total);
            updateMessage("Adding file " + (i + 1) + "/" + total);
        }

        merger.mergeDocuments(IOUtils.createMemoryOnlyStreamCache());
        log.info("Merged {} files -> {}", total, merger.getDestinationFileName());
        return Path.of(merger.getDestinationFileName());
    }

    private Path resolveUniquePath(Path target) {
        if (!Files.exists(target)) return target;
        String fileName = target.getFileName().toString();
        String name = fileName.substring(0, fileName.lastIndexOf('.'));
        String ext = fileName.substring(fileName.lastIndexOf('.'));
        Path dir = target.getParent();
        int seq = 1;
        while (Files.exists(dir.resolve(name + "(" + seq + ")" + ext))) seq++;
        return dir.resolve(name + "(" + seq + ")" + ext);
    }
}

import javafx.concurrent.Task;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class PdfConvertWorker extends Task<List<Path>> {

    private static final PluginLogger log = LoggerFactory.getLogger(PdfConvertWorker.class);

    private final List<Path> pdfPaths;
    private final Path outputDir;

    public PdfConvertWorker(List<Path> pdfPaths, Path outputDir) {
        this.pdfPaths = pdfPaths;
        this.outputDir = outputDir;
    }

    @Override
    protected List<Path> call() throws Exception {
        Files.createDirectories(outputDir);

        DocumentConverter converter = createConverter();
        List<Path> outputs = new ArrayList<>();
        int total = pdfPaths.size();

        for (int i = 0; i < total; i++) {
            Path pdf = pdfPaths.get(i);
            String pdfName = pdf.getFileName().toString();
            String docxName = pdfName.replaceAll("\\.pdf$", ".docx");
            Path outPath = outputDir.resolve(docxName);

            updateMessage("Converting " + (i + 1) + "/" + total + ": " + pdfName);
            converter.convert(pdf, outPath);
            outputs.add(outPath);
            updateProgress(i + 1, total);
        }
        return outputs;
    }

    private DocumentConverter createConverter() throws Exception {
        OfficeDetector.DetectedBackend backend = OfficeDetector.detect()
                .orElseThrow(() -> new IllegalStateException("No Office backend available"));

        return switch (backend.type()) {
            case WPS -> new fan.summer.buildintool.pdftool.converter.WpsConverter(backend.executablePath());
            case LIBRE_OFFICE, MS_WORD -> new fan.summer.buildintool.pdftool.converter.Documents4jConverter();
        };
    }
}

                resultLabel.setVisible(true);
                return;

    List<int[]> parseRanges(String input) {
        if (input == null || input.isBlank()) return List.of();
        List<int[]> ranges = new ArrayList<>();
        for (String part : input.split(",")) {
            part = part.trim();
            if (part.isEmpty()) continue;
            try {
                if (part.contains("-")) {
                    String[] s = part.split("-");
                    int start = Integer.parseInt(s[0].trim());
                    int end = Integer.parseInt(s[1].trim());
                    if (start < 1 || end < start) return null;
                    ranges.add(new int[]{start, end});
                } else {
                    int page = Integer.parseInt(part);
                    if (page < 1) return null;
                    ranges.add(new int[]{page, page});
                }
            } catch (NumberFormatException e) {
                return null;
            }
        }
        return ranges;
    }

        if (bytes < 1024 * 1024) return String.format("%.1f KB", bytes / 1024.0);

        resultLabel.setWrapText(true);
        resultLabel.setStyle("-fx-text-fill: -fx-text-secondary; -fx-font-size: 13px;");
        resultLabel.setVisible(false);

        // --- Start button ---
        startBtn.setText(I18n.get("builtin.pdf.convert.start"));
        startBtn.setStyle("-fx-background-color: linear-gradient(to bottom right, #4a9eff, #6c5ce7); " +
                "-fx-text-fill: white; -fx-font-weight: 600; -fx-background-radius: 8px; -fx-padding: 10 32;");
        startBtn.setDisable(!backendAvailable);
        startBtn.setOnAction(e -> startConvert());

        getChildren().addAll(backendStatusLabel, dropZone, fileListContainer,
                dirLabel, outputRow, progressBar, resultLabel, new HBox(startBtn));
    }

    private void detectBackend() {
        OfficeDetector.detect().ifPresentOrElse(
            backend -> {
                backendStatusLabel.setText(I18n.get("builtin.pdf.convert.backend.found", backend.displayName()));
                backendStatusLabel.setStyle("-fx-text-fill: #4cd97b; -fx-font-size: 13px; -fx-padding: 10 14; " +
                        "-fx-background-color: rgba(76,217,123,0.08); -fx-background-radius: 6px;");
                backendAvailable = true;
            },
            () -> {
                backendStatusLabel.setText(I18n.get("builtin.pdf.convert.no.backend"));
                backendStatusLabel.setStyle("-fx-text-fill: #f5a623; -fx-font-size: 13px; -fx-padding: 10 14; " +
                        "-fx-background-color: rgba(245,166,35,0.08); -fx-background-radius: 6px;");
                backendStatusLabel.setWrapText(true);
                backendAvailable = false;
            }
        );
    }

    private void selectFiles() {
        FileChooser fc = new FileChooser();
        fc.getExtensionFilters().add(new FileChooser.ExtensionFilter("PDF", "*.pdf"));
        List<File> selected = fc.showOpenMultipleDialog(getScene().getWindow());
        if (selected != null) selected.forEach(f -> addFile(f.toPath()));
    }

    private void addFile(Path path) {
        if (!path.toString().toLowerCase().endsWith(".pdf")) return;
        if (selectedFiles.contains(path)) return;
        selectedFiles.add(path);

        HBox row = new HBox(12);
        row.setAlignment(Pos.CENTER_LEFT);
        row.setPadding(new Insets(6, 10, 6, 10));
        row.setStyle("-fx-border-color: rgba(255,255,255,0.06); -fx-border-width: 0 0 1 0;");

        Label name = new Label(path.getFileName().toString());
        name.setStyle("-fx-text-fill: rgba(255,255,255,0.8); -fx-font-size: 14px;");
        HBox.setHgrow(name, Priority.ALWAYS);
        name.setMaxWidth(Double.MAX_VALUE);

        Button removeBtn = new Button("✕");
        removeBtn.setStyle("-fx-background-color: transparent; -fx-text-fill: rgba(255,80,80,0.8); -fx-padding: 4 8;");
        removeBtn.setOnAction(e -> {
            selectedFiles.remove(path);
            fileListContainer.getChildren().remove(row);
        });
        row.getChildren().addAll(name, removeBtn);
        fileListContainer.getChildren().add(row);

        if (outputDirField.getText().isEmpty()) {
            outputDirField.setText(path.getParent().toString());
        }
    }

    private void browseOutputDir() {
        DirectoryChooser dc = new DirectoryChooser();
        File d = dc.showDialog(getScene().getWindow());
        if (d != null) outputDirField.setText(d.getAbsolutePath());
    }

    private void startConvert() {
        if (selectedFiles.isEmpty() || !backendAvailable) return;
        Path outDir = Path.of(outputDirField.getText());
        startBtn.setDisable(true);
        progressBar.setVisible(true);
        progressBar.setProgress(-1); // indeterminate
        resultLabel.setVisible(false);

        PdfConvertWorker worker = new PdfConvertWorker(new ArrayList<>(selectedFiles), outDir);
        worker.setOnSucceeded(e -> Platform.runLater(() -> {
            startBtn.setDisable(false);
            progressBar.setVisible(false);
            List<Path> outputs = worker.getValue();
            resultLabel.setText(I18n.get("builtin.pdf.complete") + " (" + outputs.size() + " files)");
            resultLabel.setVisible(true);
        }));
        worker.setOnFailed(e -> Platform.runLater(() -> {
            startBtn.setDisable(false);
            progressBar.setVisible(false);
            resultLabel.setText("Error: " + worker.getException().getMessage());
            resultLabel.setVisible(true);
        }));
        new Thread(worker).start();
    }
}

    private static final PluginLogger log = LoggerFactory.getLogger(PdfSplitAiTool.class);

    @Override public String getName() { return "pdf_split"; }

    @Override public String getDescription() {
        return "Split a PDF file into multiple files by page ranges. " +
               "Args: filePath (string) — absolute path to PDF; " +
               "ranges (string) — page ranges like '1-3,5,8-10'; " +
               "outputDir (string) — output directory.";
    }

    @Override public List<AiToolParam> getParameters() {
        return List.of(
            AiToolParam.of("filePath", "string", "Absolute path to the PDF file", true),
            AiToolParam.of("ranges", "string", "Page ranges, e.g. '1-3,5,8-10'", true),
            AiToolParam.of("outputDir", "string", "Output directory for split files", true)
        );
    }

        String filePath = (String) args.get("filePath");
        String ranges = (String) args.get("ranges");
        String outputDir = (String) args.get("outputDir");

        if (filePath == null || filePath.isBlank()) return AiToolResult.error("filePath is required");
        if (ranges == null || ranges.isBlank()) return AiToolResult.error("ranges is required");
        if (outputDir == null || outputDir.isBlank()) return AiToolResult.error("outputDir is required");

        Path pdf = Path.of(filePath);
        if (!Files.exists(pdf)) return AiToolResult.error("File not found: " + filePath);

        PdfSplitPane parser = new PdfSplitPane();
        List<int[]> parsedRanges = parser.parseRanges(ranges);
        if (parsedRanges == null || parsedRanges.isEmpty()) return AiToolResult.error("Invalid ranges: " + ranges);

        try {
            fan.summer.buildintool.pdftool.worker.PdfSplitWorker worker =
                new fan.summer.buildintool.pdftool.worker.PdfSplitWorker(pdf, parsedRanges, Path.of(outputDir));
            List<Path> outputs = worker.call();
            StringBuilder sb = new StringBuilder("Split complete. Output files:\n");
            for (Path p : outputs) sb.append("- ").append(p.getFileName()).append("\n");
            log.info("AI pdf_split success: {} -> {} files", filePath, outputs.size());
            return AiToolResult.success(sb.toString());
        } catch (Exception e) {
            log.error("AI pdf_split failed: {}", e.getMessage());
            return AiToolResult.error("PDF split failed: " + e.getMessage());
        }
    }
}

import java.nio.file.Files;

public class PdfMergeAiTool implements AiTool {

    private static final PluginLogger log = LoggerFactory.getLogger(PdfMergeAiTool.class);

    @Override public String getName() { return "pdf_merge"; }

    @Override public String getDescription() {
        return "Merge multiple PDF files into one. " +
               "Args: filePaths (array of strings) — ordered list of PDF paths; " +
               "outputPath (string) — output file path for merged PDF.";
    }

    @Override public List<AiToolParam> getParameters() {
        return List.of(
            AiToolParam.of("filePaths", "array", "Ordered list of PDF file paths to merge", true),
            AiToolParam.of("outputPath", "string", "Output file path for merged PDF", true)
        );
    }

    @Override
    @SuppressWarnings("unchecked")
    public AiToolResult execute(Map<String, Object> args) {
        Object rawPaths = args.get("filePaths");
        String outputPath = (String) args.get("outputPath");

        if (rawPaths == null) return AiToolResult.error("filePaths is required");
        if (outputPath == null || outputPath.isBlank()) return AiToolResult.error("outputPath is required");

        List<String> pathStrings;
        if (rawPaths instanceof List<?> list) {
            pathStrings = list.stream().map(Object::toString).toList();
        } else {
            return AiToolResult.error("filePaths must be an array of strings");
        }

        if (pathStrings.size() < 2) return AiToolResult.error("At least 2 files required for merge");

        List<Path> paths = new ArrayList<>();
        for (String p : pathStrings) {
            Path path = Path.of(p);
            if (!Files.exists(path)) return AiToolResult.error("File not found: " + p);
            paths.add(path);
        }

        try {
            PdfMergeWorker worker = new PdfMergeWorker(paths, Path.of(outputPath));
            Path result = worker.call();
            log.info("AI pdf_merge success: {} files -> {}", paths.size(), result.getFileName());
            return AiToolResult.success("Merge complete: " + result.getFileName());
        } catch (Exception e) {
            log.error("AI pdf_merge failed: {}", e.getMessage());
            return AiToolResult.error("PDF merge failed: " + e.getMessage());
        }
    }
}
